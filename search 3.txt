    <script>
        let cachedFiles = [];
        let lastSearchResults = [];
        let sortDirection = {};
        let currentPreviewFileObj = null;

        document.getElementById('folderInput').addEventListener('change', function(e) {
            cachedFiles = Array.from(e.target.files);
            document.getElementById('status-bar').innerText = `Đã nạp ${cachedFiles.length} tệp vào bộ nhớ tạm. Sẵn sàng tìm kiếm.`;
        });

        async function executeSearch() {
            if (cachedFiles.length === 0) {
                alert("Vui lòng chọn thư mục trước khi tìm kiếm!");
                return;
            }

            const rawNameKw = document.getElementById('searchName').value.trim();
            const rawExtKw = document.getElementById('searchExt').value.toLowerCase().trim();
            const rawContentKw = document.getElementById('searchContent').value.trim();
            const caseSensitive = document.getElementById('caseSensitive').checked;
            
            const minSizeKB = parseFloat(document.getElementById('minSize').value) || 0;
            const maxSizeKB = parseFloat(document.getElementById('maxSize').value) || Infinity;

            const nameKw = caseSensitive ? rawNameKw : rawNameKw.toLowerCase();
            const contentKw = caseSensitive ? rawContentKw : rawContentKw.toLowerCase();

            let allowedExts = [];
            if (rawExtKw) {
                allowedExts = rawExtKw.split(',').map(e => e.trim().replace(/^\./, '').toLowerCase());
            }

            const statusEl = document.getElementById('status-bar');
            const tbody = document.getElementById('resultBody');
            tbody.innerHTML = `<tr><td colspan="6" class="text-center">Đang quét và phân tích dữ liệu...</td></tr>`;
            statusEl.innerText = "Đang xử lý...";

            lastSearchResults = [];

            for (let i = 0; i < cachedFiles.length; i++) {
                let file = cachedFiles[i];
                let fileName = file.name;
                let fileExt = fileName.split('.').pop().toLowerCase();
                let relativePath = file.webkitRelativePath || file.name;
                let fileSizeKB = file.size / 1024;

                if (fileSizeKB < minSizeKB || fileSizeKB > maxSizeKB) continue;

                let checkName = caseSensitive ? fileName : fileName.toLowerCase();
                let matchName = !rawNameKw || checkName.includes(nameKw);
                let matchExt = allowedExts.length === 0 || allowedExts.includes(fileExt);

                if (!matchName || !matchExt) continue;

                let matchType = [];
                if (rawNameKw) matchType.push("Tên file");
                if (allowedExts.length > 0) matchType.push("Định dạng");

                let matchContent = true;
                if (rawContentKw) {
                    if (file.size < 15 * 1024 * 1024) {
                        try {
                            let textContent = "";
                            if (['xlsx', 'xls'].includes(fileExt)) {
                                const arrayBuffer = await file.arrayBuffer();
                                const workbook = XLSX.read(arrayBuffer, { type: 'array' });
                                workbook.SheetNames.forEach(sheetName => {
                                    const sheet = workbook.Sheets[sheetName];
                                    textContent += XLSX.utils.sheet_to_csv(sheet) + "\n";
                                });
                            } else if (isTextFile(fileExt)) {
                                textContent = await file.text();
                            }

                            if (textContent) {
                                let checkContent = caseSensitive ? textContent : textContent.toLowerCase();
                                if (checkContent.includes(contentKw)) {
                                    matchType.push("Nội dung");
                                } else {
                                    matchContent = false;
                                }
                            } else {
                                matchContent = false;
                            }
                        } catch (err) {
                            matchContent = false;
                        }
                    } else {
                        matchContent = false;
                    }
                }

                if (matchName && matchExt && matchContent) {
                    lastSearchResults.push({
                        index: i, // Lưu vị trí gốc trong cachedFiles
                        fileRef: file,
                        name: fileName,
                        path: relativePath,
                        ext: fileExt,
                        rawSize: file.size,
                        size: formatBytes(file.size),
                        match: matchType.join(" + ") || "Khớp điều kiện"
                    });
                }
            }

            renderTable(lastSearchResults);
            statusEl.innerText = `Tìm thấy ${lastSearchResults.length} kết quả phù hợp trên tổng số ${cachedFiles.length} tệp.`;
        }

        function renderTable(results) {
            const tbody = document.getElementById('resultBody');
            tbody.innerHTML = '';
            if (results.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="text-center">Không tìm thấy tệp nào phù hợp với điều kiện.</td></tr>`;
                return;
            }
            results.forEach((r, idx) => {
                tbody.innerHTML += `
                    <tr>
                        <td><strong>${escapeHtml(r.name)}</strong></td>
                        <td><small style="color: #666;">${escapeHtml(r.path)}</small></td>
                        <td><span class="badge">.${escapeHtml(r.ext)}</span></td>
                        <td>${r.size}</td>
                        <td><span class="match-content">${r.match}</span></td>
                        <td class="text-center">
                            <button class="btn btn-sm" onclick="previewFile(${idx})">Xem</button>
                            <button class="btn btn-sec btn-sm" onclick="downloadFile(${idx})">Tải</button>
                        </td>
                    </tr>
                `;
            });
        }

        async function previewFile(resultIndex) {
            const item = lastSearchResults[resultIndex];
            if (!item) return;
            currentPreviewFileObj = item.fileRef;

            const modal = document.getElementById('previewModal');
            const modalTitle = document.getElementById('modalTitle');
            const modalBody = document.getElementById('modalBody');

            modalTitle.innerText = `Xem trước: ${item.name}`;
            modalBody.innerText = "Đang đọc nội dung tệp...";
            modal.style.display = "flex";

            try {
                let file = item.fileRef;
                let ext = item.ext;

                if (['xlsx', 'xls'].includes(ext)) {
                    const arrayBuffer = await file.arrayBuffer();
                    const workbook = XLSX.read(arrayBuffer, { type: 'array' });
                    let htmlContent = "";
                    workbook.SheetNames.forEach(sheetName => {
                        const sheet = workbook.Sheets[sheetName];
                        htmlContent += `<h4>Sheet: ${sheetName}</h4>`;
                        htmlContent += XLSX.utils.sheet_to_html(sheet, { header: "" }) + "<br><hr><br>";
                    });
                    modalBody.innerHTML = htmlContent;
                } else if (isTextFile(ext)) {
                    const text = await file.text();
                    modalBody.innerText = text;
                } else {
                    modalBody.innerText = `Định dạng tệp .${ext} không hỗ trợ xem trước trực tiếp văn bản.\nBạn có thể bấm nút "Tải tệp này xuống" ở dưới để mở bằng phần mềm trên máy tính.`;
                }
            } catch (err) {
                modalBody.innerText = "Lỗi khi đọc nội dung tệp: " + err.message;
            }
        }

        function downloadFile(resultIndex) {
            const item = lastSearchResults[resultIndex];
            if (!item) return;
            triggerFileDownload(item.fileRef, item.name);
        }

        function downloadCurrentPreviewFile() {
            if (!currentPreviewFileObj) return;
            triggerFileDownload(currentPreviewFileObj, currentPreviewFileObj.name);
        }

        function triggerFileDownload(fileObj, filename) {
            const url = URL.createObjectURL(fileObj);
            const a = document.createElement('a');
            a.href = url;
            a.download = filename;
            a.click();
            URL.revokeObjectURL(url);
        }

        function closeModal() {
            document.getElementById('previewModal').style.display = "none";
            currentPreviewFileObj = null;
        }

        function sortTable(column) {
            if (lastSearchResults.length === 0) return;
            if (!sortDirection[column]) sortDirection[column] = 'asc';
            else sortDirection[column] = sortDirection[column] === 'asc' ? 'desc' : 'asc';

            const dir = sortDirection[column];
            lastSearchResults.sort((a, b) => {
                let valA = a[column];
                let valB = b[column];
                if (typeof valA === 'number' && typeof valB === 'number') {
                    return dir === 'asc' ? valA - valB : valB - valA;
                }
                valA = String(valA).toLowerCase();
                valB = String(valB).toLowerCase();
                if (valA < valB) return dir === 'asc' ? -1 : 1;
                if (valA > valB) return dir === 'asc' ? 1 : -1;
                return 0;
            });
            renderTable(lastSearchResults);
        }

        function exportResultsCSV() {
            if (lastSearchResults.length === 0) {
                alert("Không có kết quả tìm kiếm để xuất!");
                return;
            }
            let csv = "FileName,Path,Extension,Size,MatchType\n";
            lastSearchResults.forEach(r => {
                csv += `"${r.name}","${r.path}","${r.ext}","${r.size}","${r.match}"\n`;
            });
            const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url; a.download = `FileSearch_Results_${Date.now()}.csv`;
            a.click();
            URL.revokeObjectURL(url);
        }

        function isTextFile(ext) {
            const textExts = ['txt', 'csv', 'log', 'json', 'html', 'htm', 'xml', 'js', 'css', 'md', 'bat', 'ini', 'cfg'];
            return textExts.includes(ext);
        }

        function formatBytes(bytes, decimals = 2) {
            if (bytes === 0) return '0 Bytes';
            const k = 1024;
            const dm = decimals < 0 ? 0 : decimals;
            const sizes = ['Bytes', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
        }

        function escapeHtml(str) {
            return String(str).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }

        function resetForm() {
            document.getElementById('folderInput').value = '';
            document.getElementById('searchName').value = '';
            document.getElementById('searchExt').value = '';
            document.getElementById('searchContent').value = '';
            document.getElementById('minSize').value = '';
            document.getElementById('maxSize').value = '';
            document.getElementById('caseSensitive').checked = false;
            document.getElementById('resultBody').innerHTML = `<tr><td colspan="6" class="text-center">Vui lòng chọn thư mục và nhập từ khóa tìm kiếm.</td></tr>`;
            document.getElementById('status-bar').innerText = "Đã làm mới bộ lọc.";
            cachedFiles = [];
            lastSearchResults = [];
        }
    </script>
