  <script>
    // State management
    let filesArray = [];
    let detectedColumns = [];
    let lastMergedRows = [];

    // --- LOGGING ---
    function addLog(msg, type = 'info') {
      const consoleEl = document.getElementById('logConsole');
      const time = new Date().toLocaleTimeString();
      let colorClass = 'text-slate-400';
      if (type === 'success') colorClass = 'text-emerald-400';
      if (type === 'warn') colorClass = 'text-amber-400';
      if (type === 'error') colorClass = 'text-rose-400';

      const line = document.createElement('div');
      line.className = colorClass;
      line.innerHTML = `<span class="text-slate-600">[${time}]</span> ${msg}`;
      consoleEl.appendChild(line);
      consoleEl.scrollTop = consoleEl.scrollHeight;
    }

    function clearLogs() {
      document.getElementById('logConsole').innerHTML = '<div class="text-slate-500">[System]: Nhật ký đã được làm sạch.</div>';
    }

    // --- FILE HANDLING ---
    function handleFilesSelect(e) {
      const selectedFiles = Array.from(e.target.files);
      addFiles(selectedFiles);
    }

    // Drag and Drop
    const dropZone = document.getElementById('dropZone');
    dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('border-emerald-500'); });
    dropZone.addEventListener('dragleave', () => dropZone.classList.remove('border-emerald-500'));
    dropZone.addEventListener('drop', (e) => {
      e.preventDefault();
      dropZone.classList.remove('border-emerald-500');
      if (e.dataTransfer.files.length > 0) {
        addFiles(Array.from(e.dataTransfer.files));
      }
    });

    function addFiles(files) {
      files.forEach(file => {
        if (!filesArray.some(f => f.name === file.name && f.size === file.size)) {
          filesArray.push(file);
          addLog(`Đã thêm file: ${file.name}`, 'info');
        }
      });
      updateFileListUI();
      scanColumnsFromFiles();
    }

    function removeFile(index) {
      const name = filesArray[index].name;
      filesArray.splice(index, 1);
      addLog(`Đã xóa file: ${name}`, 'warn');
      updateFileListUI();
      scanColumnsFromFiles();
    }

    function removeAllFiles() {
      filesArray = [];
      detectedColumns = [];
      lastMergedRows = [];
      addLog('Đã xóa tất cả các file.', 'warn');
      updateFileListUI();
      renderColumnSelection();
      renderPreview([]);
    }

    function moveFile(index, direction) {
      const targetIndex = index + direction;
      if (targetIndex < 0 || targetIndex >= filesArray.length) return;
      const temp = filesArray[index];
      filesArray[index] = filesArray[targetIndex];
      filesArray[targetIndex] = temp;
      updateFileListUI();
    }

    function updateFileListUI() {
      const fileListEl = document.getElementById('fileList');
      const fileCountEl = document.getElementById('fileCount');
      const totalSizeEl = document.getElementById('totalSize');

      fileCountEl.textContent = filesArray.length;
      let totalSizeBytes = filesArray.reduce((acc, f) => acc + f.size, 0);
      totalSizeEl.textContent = (totalSizeBytes / (1024 * 1024)).toFixed(2) + ' MB';

      if (filesArray.length === 0) {
        fileListEl.innerHTML = '<p class="text-xs text-slate-500 text-center py-6 border border-slate-800/60 rounded-xl">Chưa có file nào được nạp.</p>';
        return;
      }

      fileListEl.innerHTML = filesArray.map((file, idx) => `
        <div class="flex items-center justify-between p-3 rounded-xl bg-slate-900/60 border border-slate-800 hover:border-slate-700 transition">
          <div class="flex items-center gap-3 truncate mr-2">
            <i class="fa-solid fa-file-excel text-emerald-400"></i>
            <div class="truncate">
              <p class="text-xs font-medium text-slate-200 truncate">${file.name}</p>
              <p class="text-[10px] text-slate-500">${(file.size / 1024).toFixed(1)} KB</p>
            </div>
          </div>
          <div class="flex items-center gap-1.5 shrink-0">
            <button onclick="moveFile(${idx}, -1)" ${idx === 0 ? 'disabled' : ''} class="w-6 h-6 rounded bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-slate-300 text-xs flex items-center justify-center">
              <i class="fa-solid fa-chevron-up"></i>
            </button>
            <button onclick="moveFile(${idx}, 1)" ${idx === filesArray.length - 1 ? 'disabled' : ''} class="w-6 h-6 rounded bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-slate-300 text-xs flex items-center justify-center">
              <i class="fa-solid fa-chevron-down"></i>
            </button>
            <button onclick="removeFile(${idx})" class="w-6 h-6 rounded bg-rose-500/10 hover:bg-rose-500/20 text-rose-400 text-xs flex items-center justify-center ml-1">
              <i class="fa-solid fa-xmark"></i>
            </button>
          </div>
        </div>
      `).join('');
    }

    // --- COLUMN SCANNING ---
    async function scanColumnsFromFiles() {
      if (filesArray.length === 0) {
        detectedColumns = [];
        renderColumnSelection();
        return;
      }

      const colSet = new Set();

      for (let file of filesArray) {
        try {
          const buffer = await file.arrayBuffer();
          const workbook = XLSX.read(buffer, { type: 'array' });
          const firstSheetName = workbook.SheetNames[0];
          const sheet = workbook.Sheets[firstSheetName];
          const json = XLSX.utils.sheet_to_json(sheet, { header: 1 });
          if (json.length > 0) {
            const headers = json[0];
            if (Array.isArray(headers)) {
              headers.forEach(h => {
                if (h !== undefined && h !== null) colSet.add(String(h).trim());
              });
            }
          }
        } catch (err) {
          addLog(`Lỗi đọc tiêu đề cột file ${file.name}: ${err.message}`, 'error');
        }
      }

      detectedColumns = Array.from(colSet);
      renderColumnSelection();
    }

    function renderColumnSelection() {
      const container = document.getElementById('columnSelectionContainer');
      if (detectedColumns.length === 0) {
        container.innerHTML = '<p class="text-xs text-slate-500 col-span-2 text-center py-6 border border-slate-800/60 rounded-xl">Chưa tìm thấy tiêu đề cột nào.</p>';
        return;
      }

      container.innerHTML = detectedColumns.map((col, idx) => `
        <div class="flex items-center gap-2 p-2 rounded-xl bg-slate-900/50 border border-slate-800">
          <input type="checkbox" id="col_chk_${idx}" value="${col}" checked class="col-checkbox rounded border-slate-700 bg-slate-800 text-emerald-500 focus:ring-emerald-500">
          <span class="text-xs text-slate-300 font-medium truncate w-1/3" title="${col}">${col}</span>
          <i class="fa-solid fa-arrow-right text-[10px] text-slate-600"></i>
          <input type="text" id="col_ren_${idx}" value="${col}" placeholder="Tên cột mới" class="flex-1 bg-slate-950 border border-slate-800 rounded-lg px-2 py-1 text-xs text-slate-200 focus:outline-none focus:border-emerald-500">
        </div>
      `).join('');
    }

    function selectAllColumns(select) {
      const checkboxes = document.querySelectorAll('.col-checkbox');
      checkboxes.forEach(cb => cb.checked = select);
      addLog(select ? 'Đã chọn tất cả các cột.' : 'Đã bỏ chọn tất cả các cột.', 'info');
    }

    // --- PROCESSING & MERGING ---
    async function processAndMerge() {
      if (filesArray.length === 0) {
        alert('Vui lòng nạp ít nhất 1 file Excel hoặc CSV!');
        addLog('Chưa có file để xử lý.', 'error');
        return;
      }

      const progressContainer = document.getElementById('progressContainer');
      const progressBar = document.getElementById('progressBar');
      const progressText = document.getElementById('progressText');
      const progressPercent = document.getElementById('progressPercent');

      progressContainer.classList.remove('hidden');
      progressBar.style.width = '0%';

      const mode = document.querySelector('input[name="mergeMode"]:checked').value;
      const addSrcFile = document.getElementById('addSourceFile').checked;
      const addSrcSheet = document.getElementById('addSourceSheet').checked;
      const autoFit = document.getElementById('autoFitCols').checked;

      // Prepare Column Mappings
      const selectedColMappings = [];
      const checkboxes = document.querySelectorAll('.col-checkbox');
      checkboxes.forEach((cb, idx) => {
        if (cb.checked) {
          const originalName = cb.value;
          const newName = document.getElementById(`col_ren_${idx}`).value.trim() || originalName;
          selectedColMappings.push({ original: originalName, renamed: newName });
        }
      });

      if (mode === 'append' && selectedColMappings.length === 0) {
        alert('Vui lòng chọn ít nhất 1 cột để xuất!');
        addLog('Chưa chọn cột nào.', 'error');
        progressContainer.classList.add('hidden');
        return;
      }

      try {
        addLog('Bắt đầu gộp file...', 'info');
        const mergedWorkbook = XLSX.utils.book_new();
        let allMergedRows = [];

        for (let i = 0; i < filesArray.length; i++) {
          const file = filesArray[i];
          const percent = Math.round(((i + 1) / filesArray.length) * 100);
          progressBar.style.width = percent + '%';
          progressPercent.textContent = percent + '%';
          progressText.textContent = `Đang đọc (${i + 1}/${filesArray.length}): ${file.name}`;

          const buffer = await file.arrayBuffer();
          const wb = XLSX.read(buffer, { type: 'array' });

          if (mode === 'append') {
            wb.SheetNames.forEach(sheetName => {
              const sheet = wb.Sheets[sheetName];
              const jsonRows = XLSX.utils.sheet_to_json(sheet, { defval: "" });

              jsonRows.forEach(row => {
                const formattedRow = {};
                
                if (addSrcFile) formattedRow['File Nguồn'] = file.name;
                if (addSrcSheet) formattedRow['Sheet Nguồn'] = sheetName;

                selectedColMappings.forEach(col => {
                  formattedRow[col.renamed] = row[col.original] !== undefined ? row[col.original] : "";
                });

                allMergedRows.push(formattedRow);
              });
            });
          } else {
            // Mode: Multiple Sheets
            wb.SheetNames.forEach((sheetName) => {
              let safeSheetName = `${file.name.replace(/\.[^/.]+$/, "")}_${sheetName}`.substring(0, 30);
              const sheet = wb.Sheets[sheetName];
              XLSX.utils.book_append_sheet(mergedWorkbook, sheet, safeSheetName);
            });
          }
        }

        if (mode === 'append') {
          lastMergedRows = allMergedRows;
          const mergedSheet = XLSX.utils.json_to_sheet(allMergedRows);

          // Auto-fit Column Widths
          if (autoFit && allMergedRows.length > 0) {
            const keys = Object.keys(allMergedRows[0]);
            const colWidths = keys.map(key => {
              let maxLen = key.toString().length;
              allMergedRows.slice(0, 100).forEach(r => {
                const val = r[key] ? r[key].toString() : '';
                if (val.length > maxLen) maxLen = val.length;
              });
              return { wch: Math.min(Math.max(maxLen + 3, 10), 60) };
            });
            mergedSheet['!cols'] = colWidths;
          }

          XLSX.utils.book_append_sheet(mergedWorkbook, mergedSheet, "Data_Merged");
          renderPreview(allMergedRows);
        }

        // TRIGGER DOWNLOAD USING BLOB URL
        const filename = (document.getElementById('outputFileName').value.trim() || 'Data_Merged_File') + '.xlsx';
        const wbout = XLSX.write(mergedWorkbook, { bookType: 'xlsx', type: 'array' });
        const blob = new Blob([wbout], { type: 'application/octet-stream' });

        const blobUrl = URL.createObjectURL(blob);
        const downloadLink = document.createElement('a');
        downloadLink.href = blobUrl;
        downloadLink.download = filename;
        document.body.appendChild(downloadLink);
        downloadLink.click();

        setTimeout(() => {
          document.body.removeChild(downloadLink);
          URL.revokeObjectURL(blobUrl);
        }, 1000);

        addLog(`Gộp & Xuất file thành công: ${filename}`, 'success');
      } catch (err) {
        addLog(`Lỗi xử lý file: ${err.message}`, 'error');
        alert('Đã xảy ra lỗi trong quá trình xử lý file: ' + err.message);
      } finally {
        setTimeout(() => {
          progressContainer.classList.add('hidden');
        }, 800);
      }
    }

    // --- RENDER PREVIEW TABLE ---
    function renderPreview(rows) {
      const thead = document.getElementById('previewThead');
      const tbody = document.getElementById('previewTbody');
      const previewCount = document.getElementById('previewCount');

      previewCount.textContent = rows.length + ' dòng';

      if (!rows || rows.length === 0) {
        thead.innerHTML = '<tr><th class="p-4 text-center text-slate-500">Chưa có dữ liệu preview</th></tr>';
        tbody.innerHTML = '';
        return;
      }

      const keys = Object.keys(rows[0]);
      thead.innerHTML = `<tr>${keys.map(k => `<th class="p-3 text-slate-300 font-semibold border-b border-slate-800">${k}</th>`).join('')}</tr>`;

      const previewRows = rows.slice(0, 50);
      tbody.innerHTML = previewRows.map(row => `
        <tr class="hover:bg-slate-900/40 transition">
          ${keys.map(k => `<td class="p-3 border-b border-slate-800/40 truncate max-w-xs">${row[k] !== undefined ? row[k] : ''}</td>`).join('')}
        </tr>
      `).join('');
    }
