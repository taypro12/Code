  <script>
    let filesArray = [];
    let detectedColumns = [];

    // LOGGING
    function addLog(msg, type = 'info') {
      const consoleEl = document.getElementById('logConsole');
      const time = new Date().toLocaleTimeString();
      let colorClass = 'text-slate-400';
      if (type === 'success') colorClass = 'text-cyan-400';
      if (type === 'warn') colorClass = 'text-amber-400';
      if (type === 'error') colorClass = 'text-rose-400';

      const line = document.createElement('div');
      line.className = colorClass;
      line.innerHTML = `<span class="text-slate-600">[${time}]</span> ${msg}`;
      consoleEl.appendChild(line);
      consoleEl.scrollTop = consoleEl.scrollHeight;
    }

    function clearLogs() {
      document.getElementById('logConsole').innerHTML = '<div class="text-slate-500">[System]: Nhật ký đã xóa.</div>';
    }

    // FILE HANDLING
    function handleFilesSelect(e) {
      addFiles(Array.from(e.target.files));
    }

    const dropZone = document.getElementById('dropZone');
    dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('border-cyan-500'); });
    dropZone.addEventListener('dragleave', () => dropZone.classList.remove('border-cyan-500'));
    dropZone.addEventListener('drop', (e) => {
      e.preventDefault();
      dropZone.classList.remove('border-cyan-500');
      if (e.dataTransfer.files.length > 0) addFiles(Array.from(e.dataTransfer.files));
    });

    function addFiles(files) {
      files.forEach(file => {
        if (!filesArray.some(f => f.name === file.name && f.size === file.size)) {
          filesArray.push(file);
          addLog(`Đã nạp file: ${file.name}`, 'info');
        }
      });
      updateFileListUI();
      scanColumnsFromFiles();
    }

    function removeFile(index) {
      const name = filesArray[index].name;
      filesArray.splice(index, 1);
      addLog(`Đã xóa file: ${name}`, 'warn');
      updateFileListUI();
      scanColumnsFromFiles();
    }

    function removeAllFiles() {
      filesArray = [];
      detectedColumns = [];
      addLog('Đã xóa tất cả file.', 'warn');
      updateFileListUI();
      renderColumnSelection();
      renderPreview([]);
    }

    function moveFile(index, dir) {
      const targetIndex = index + dir;
      if (targetIndex < 0 || targetIndex >= filesArray.length) return;
      const temp = filesArray[index];
      filesArray[index] = filesArray[targetIndex];
      filesArray[targetIndex] = temp;
      updateFileListUI();
    }

    function updateFileListUI() {
      const fileListEl = document.getElementById('fileList');
      document.getElementById('fileCount').textContent = filesArray.length;
      let totalSizeBytes = filesArray.reduce((acc, f) => acc + f.size, 0);
      document.getElementById('totalSize').textContent = (totalSizeBytes / 1024).toFixed(1) + ' KB';

      if (filesArray.length === 0) {
        fileListEl.innerHTML = '<p class="text-xs text-slate-500 text-center py-6 border border-slate-800/60 rounded-xl">Chưa có file nào được nạp.</p>';
        return;
      }

      fileListEl.innerHTML = filesArray.map((file, idx) => `
        <div class="flex items-center justify-between p-3 rounded-xl bg-slate-900/60 border border-slate-800">
          <div class="flex items-center gap-3 truncate mr-2">
            <i class="fa-solid fa-file-lines text-cyan-400"></i>
            <div class="truncate">
              <p class="text-xs font-medium text-slate-200 truncate">${file.name}</p>
              <p class="text-[10px] text-slate-500">${(file.size / 1024).toFixed(1)} KB</p>
            </div>
          </div>
          <div class="flex items-center gap-1.5 shrink-0">
            <button onclick="moveFile(${idx}, -1)" ${idx === 0 ? 'disabled' : ''} class="w-6 h-6 rounded bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-slate-300 text-xs flex items-center justify-center">
              <i class="fa-solid fa-chevron-up"></i>
            </button>
            <button onclick="moveFile(${idx}, 1)" ${idx === filesArray.length - 1 ? 'disabled' : ''} class="w-6 h-6 rounded bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-slate-300 text-xs flex items-center justify-center">
              <i class="fa-solid fa-chevron-down"></i>
            </button>
            <button onclick="removeFile(${idx})" class="w-6 h-6 rounded bg-rose-500/10 hover:bg-rose-500/20 text-rose-400 text-xs flex items-center justify-center ml-1">
              <i class="fa-solid fa-xmark"></i>
            </button>
          </div>
        </div>
      `).join('');
    }

    // DELIMITER AUTO-DETECTION
    function detectDelimiter(text) {
      const sample = text.slice(0, 5000);
      const counts = {
        '\t': (sample.match(/\t/g) || []).length,
        ',': (sample.match(/,/g) || []).length,
        ';': (sample.match(/;/g) || []).length,
        '|': (sample.match(/\|/g) || []).length,
      };
      let best = '\t';
      let max = -1;
      for (let k in counts) {
        if (counts[k] > max) { max = counts[k]; best = k; }
      }
      return best;
    }

    function getSelectedDelimiter(textSample) {
      const val = document.getElementById('delimiterSelect').value;
      if (val === 'tab') return '\t';
      if (val === 'comma') return ',';
      if (val === 'semicolon') return ';';
      if (val === 'pipe') return '|';
      if (val === 'space') return ' ';
      return detectDelimiter(textSample);
    }

    // SCAN COLUMNS
    async function scanColumnsFromFiles() {
      if (filesArray.length === 0) {
        detectedColumns = [];
        renderColumnSelection();
        return;
      }

      const colSet = new Set();
      const encoding = document.getElementById('encodingSelect').value;
      const skipRows = parseInt(document.getElementById('skipRowsInput').value) || 0;
      const hasHeader = document.getElementById('hasHeaderChk').checked;

      for (let file of filesArray) {
        try {
          const buffer = await file.arrayBuffer();
          const decoder = new TextDecoder(encoding);
          const text = decoder.decode(buffer);
          
          let lines = text.split(/\r?\n/).filter(line => line.trim() !== '');
          lines = lines.slice(skipRows);
          
          if (lines.length > 0) {
            const delim = getSelectedDelimiter(lines[0]);
            const firstLineCols = lines[0].split(delim).map(c => c.trim().replace(/^["']|["']$/g, ''));
            
            if (hasHeader) {
              firstLineCols.forEach(col => colSet.add(col));
            } else {
              firstLineCols.forEach((_, i) => colSet.add(`Cot_${i + 1}`));
            }
          }
        } catch (err) {
          addLog(`Lỗi đọc file ${file.name}: ${err.message}`, 'error');
        }
      }

      detectedColumns = Array.from(colSet);
      renderColumnSelection();
    }

    function renderColumnSelection() {
      const container = document.getElementById('columnSelectionContainer');
      if (detectedColumns.length === 0) {
        container.innerHTML = '<p class="text-xs text-slate-500 col-span-2 text-center py-6 border border-slate-800/60 rounded-xl">Chưa tìm thấy tiêu đề cột.</p>';
        return;
      }

      container.innerHTML = detectedColumns.map((col, idx) => `
        <div class="flex items-center gap-2 p-2 rounded-xl bg-slate-900/50 border border-slate-800">
          <input type="checkbox" id="col_chk_${idx}" value="${col}" checked class="col-checkbox rounded border-slate-700 bg-slate-800 text-cyan-500 focus:ring-cyan-500">
          <span class="text-xs text-slate-300 font-medium truncate w-1/3" title="${col}">${col}</span>
          <i class="fa-solid fa-arrow-right text-[10px] text-slate-600"></i>
          <input type="text" id="col_ren_${idx}" value="${col}" class="flex-1 bg-slate-950 border border-slate-800 rounded-lg px-2 py-1 text-xs text-slate-200 focus:outline-none focus:border-cyan-500">
        </div>
      `).join('');
    }

    function selectAllColumns(select) {
      document.querySelectorAll('.col-checkbox').forEach(cb => cb.checked = select);
    }

    // PROCESS & EXPORT
    async function processAndExport() {
      if (filesArray.length === 0) {
        alert('Vui lòng nạp ít nhất 1 file TXT!');
        return;
      }

      const encoding = document.getElementById('encodingSelect').value;
      const skipRows = parseInt(document.getElementById('skipRowsInput').value) || 0;
      const hasHeader = document.getElementById('hasHeaderChk').checked;
      const addSrcFile = document.getElementById('addSourceFileChk').checked;
      const autoFit = document.getElementById('autoFitColsChk').checked;
      const format = document.getElementById('exportFormatSelect').value;

      // Map selected columns
      const selectedCols = [];
      document.querySelectorAll('.col-checkbox').forEach((cb, idx) => {
        if (cb.checked) {
          const original = cb.value;
          const renamed = document.getElementById(`col_ren_${idx}`).value.trim() || original;
          selectedCols.push({ original, renamed, idx });
        }
      });

      if (selectedCols.length === 0) {
        alert('Vui lòng chọn ít nhất 1 cột để xuất!');
        return;
      }

      try {
        addLog('Đang tiến hành gộp dữ liệu...', 'info');
        let allRows = [];

        for (let file of filesArray) {
          const buffer = await file.arrayBuffer();
          const decoder = new TextDecoder(encoding);
          const text = decoder.decode(buffer);
          
          let lines = text.split(/\r?\n/).filter(line => line.trim() !== '');
          lines = lines.slice(skipRows);
          if (lines.length === 0) continue;

          const delim = getSelectedDelimiter(lines[0]);
          let headers = [];

          if (hasHeader) {
            headers = lines[0].split(delim).map(c => c.trim().replace(/^["']|["']$/g, ''));
            lines = lines.slice(1);
          } else {
            const firstCols = lines[0].split(delim);
            headers = firstCols.map((_, i) => `Cot_${i + 1}`);
          }

          lines.forEach(line => {
            const cols = line.split(delim).map(c => c.trim().replace(/^["']|["']$/g, ''));
            const rowObj = {};

            if (addSrcFile) rowObj['Tên File Nguồn'] = file.name;

            selectedCols.forEach(sc => {
              let val = '';
              const colIdx = headers.indexOf(sc.original);
              if (colIdx !== -1 && cols[colIdx] !== undefined) {
                val = cols[colIdx];
              }
              rowObj[sc.renamed] = val;
            });

            allRows.push(rowObj);
          });
        }

        renderPreview(allRows);

        // Build Excel Sheet
        const ws = XLSX.utils.json_to_sheet(allRows);

        if (autoFit && allRows.length > 0) {
          const keys = Object.keys(allRows[0]);
          ws['!cols'] = keys.map(k => {
            let maxLen = k.length;
            allRows.slice(0, 100).forEach(r => {
              const len = r[k] ? r[k].toString().length : 0;
              if (len > maxLen) maxLen = len;
            });
            return { wch: Math.min(Math.max(maxLen + 3, 10), 60) };
          });
        }

        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, ws, "TXT_Merged_Data");

        // Set BookType: xlsm or xlsx
        const bookType = format === 'xlsm' ? 'xlsm' : 'xlsx';
        const ext = '.' + bookType;
        const filename = (document.getElementById('outputFileName').value.trim() || 'Merged_Data_Result') + ext;

        const wbout = XLSX.write(wb, { bookType: bookType, type: 'array' });
        const blob = new Blob([wbout], { type: 'application/octet-stream' });

        const blobUrl = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = blobUrl;
        a.download = filename;
        document.body.appendChild(a);
        a.click();

        setTimeout(() => {
          document.body.removeChild(a);
          URL.revokeObjectURL(blobUrl);
        }, 1000);

        addLog(`Xuất thành công file ${filename} (${allRows.length} dòng)`, 'success');
      } catch (err) {
        addLog(`Lỗi xử lý gộp file: ${err.message}`, 'error');
        alert('Lỗi: ' + err.message);
      }
    }

    // PREVIEW
    function renderPreview(rows) {
      const thead = document.getElementById('previewThead');
      const tbody = document.getElementById('previewTbody');
      document.getElementById('previewCount').textContent = rows.length + ' dòng';

      if (!rows || rows.length === 0) {
        thead.innerHTML = '<tr><th class="p-4 text-center text-slate-500">Chưa có dữ liệu preview</th></tr>';
        tbody.innerHTML = '';
        return;
      }

      const keys = Object.keys(rows[0]);
      thead.innerHTML = `<tr>${keys.map(k => `<th class="p-3 text-slate-300 font-semibold border-b border-slate-800">${k}</th>`).join('')}</tr>`;

      tbody.innerHTML = rows.slice(0, 50).map(row => `
        <tr class="hover:bg-slate-900/40 transition">
          ${keys.map(k => `<td class="p-3 border-b border-slate-800/40 truncate max-w-xs">${row[k] !== undefined ? row[k] : ''}</td>`).join('')}
        </tr>
      `).join('');
    }

    // SAVE HTML APP
    function downloadSelfHtml() {
      const htmlContent = "<!DOCTYPE html>\n" + document.documentElement.outerHTML;
      const blob = new Blob([htmlContent], { type: 'text/html' });
      const blobUrl = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = blobUrl;
      a.download = 'txt_to_xlsm_merger.html';
      document.body.appendChild(a);
      a.click();
      setTimeout(() => {
        document.body.removeChild(a);
        URL.revokeObjectURL(blobUrl);
      }, 500);
      addLog('Đã lưu ứng dụng về máy thành công.', 'success');
    }
