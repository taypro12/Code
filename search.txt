        let sapData = [];

        function showNotif(msg) {
            const n = document.getElementById('notif');
            n.innerText = msg; n.style.display = 'block';

        }

        function switchTab(tabId, el) {

            document.querySelectorAll('.menu-item').forEach(m => m.classList.remove('active'));

        // --- MODULE: SAP DATA PARSER (Bảo lưu số 0 ở đầu) ---
        function loadSAPFile(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    const text = evt.target.result;
                    const lines = text.split(/\r\n|\n/);
                    if (lines.length === 0) return;
                    
                    const firstLine = lines[0];
                    let sep = ',';
                    if (firstLine.includes('\t')) sep = '\t';
                    else if (firstLine.includes(';')) sep = ';';

                    sapData = [];
                    for (let i = 1; i < lines.length; i++) {
                        if (!lines[i].trim()) continue;
                        const cols = lines[i].split(sep).map(c => c.trim().replace(/^"|"$/g, ''));
                        if (cols.length >= 2) {
                            sapData.push({
                                Material: String(cols[0] || '').trim(),
                                Description: String(cols[1] || '').trim(),
                                PartNumber: String(cols[2] || '').trim(),
                                Location: String(cols[3] || '').trim(),
                                Quantity: parseFloat(cols[4]) || 0,
                                Unit: String(cols[5] || '').trim(),
                                Plant: String(cols[6] || '').trim()
                            });
                        }
                    }

                    filteredData = [...sapData];
                    renderSAPTable();
                    updateDashboard();
                    showNotif(`Đã tải thành công ${sapData.length} dòng dữ liệu SAP!`);
                } catch (err) {
                    showNotif("Lỗi đọc tệp SAP: " + err.message);
                }
            };
            reader.readAsText(file, 'UTF-8');
        }

        function renderSAPTable() {
            const tbody = document.querySelector('#sap-table tbody');
            tbody.innerHTML = '';
            const display = filteredData.slice(0, 150);
            if (display.length === 0) {
                tbody.innerHTML = '<tr><td colspan="7" class="text-center">Không có dữ liệu hiển thị</td></tr>';
                return;
            }
            display.forEach(r => {
                tbody.innerHTML += `<tr>
                    <td><strong>${escapeHtml(r.Material)}</strong></td>
                    <td>${escapeHtml(r.Description)}</td>
                    <td>${escapeHtml(r.PartNumber)}</td>
                    <td>${escapeHtml(r.Location)}</td>
                    <td class="${r.Quantity < 0 ? 'text-danger' : ''}">${r.Quantity.toLocaleString('vi-VN')}</td>
                    <td>${escapeHtml(r.Unit)}</td>
                    <td>${escapeHtml(r.Plant)}</td>
                </tr>`;
            });
            document.getElementById('sap-count').innerText = `Hiển thị ${display.length} / Tổng số ${filteredData.length} dòng`;
        }

        function filterSAP() {
            const type = document.getElementById('sap-filter').value;
            const kw = document.getElementById('sap-search').value.toLowerCase();
            filteredData = sapData.filter(r => {
                let matchQ = true;
                if (type === 'gt0') matchQ = r.Quantity > 0;
                else if (type === 'eq0') matchQ = r.Quantity === 0;
                else if (type === 'lt0') matchQ = r.Quantity < 0;

                let matchK = !kw || r.Material.toLowerCase().includes(kw) || r.Description.toLowerCase().includes(kw) || r.Location.toLowerCase().includes(kw);
                return matchQ && matchK;
            });
            renderSAPTable();
        }

        function updateDashboard() {
            const total = sapData.length;
            const instock = sapData.filter(r => r.Quantity > 0).length;
            const outstock = sapData.filter(r => r.Quantity === 0).length;
            const mats = sapData.map(r => r.Material);
            const dups = mats.filter((m, i) => mats.indexOf(m) !== i).length;

            document.getElementById('stat-total').innerText = total.toLocaleString();
            document.getElementById('stat-instock').innerText = instock.toLocaleString();
            document.getElementById('stat-outstock').innerText = outstock.toLocaleString();
            document.getElementById('stat-dup').innerText = dups.toLocaleString();
        }

        function exportSAPCSV() {
            if (filteredData.length === 0) { showNotif("Không có dữ liệu xuất!"); return; }
            let csv = "Material,Description,PartNumber,Location,Quantity,Unit,Plant\n";
            filteredData.forEach(r => {
                csv += `"${r.Material}","${r.Description}","${r.PartNumber}","${r.Location}",${r.Quantity},"${r.Unit}","${r.Plant}"\n`;
            });
            downloadBlob(csv, `SAP_Export_${Date.now()}.csv`, 'text/csv;charset=utf-8;');
        }

        // --- MODULE: COMPARE INVENTORY ---
        async function runComparison() {
            const fileA = document.getElementById('comp-file-a').files[0];
            const fileB = document.getElementById('comp-file-b').files[0];
            if (!fileA || !fileB) { showNotif("Vui lòng chọn đủ 2 tệp Kỳ Cũ và Kỳ Mới!"); return; }

            const parseFileAsync = (file) => new Promise((resolve) => {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const lines = e.target.result.split(/\r\n|\n/);
                    let map = new Map();
                    let sep = lines[0].includes('\t') ? '\t' : (lines[0].includes(';') ? ';' : ',');
                    for (let i = 1; i < lines.length; i++) {
                        if (!lines[i].trim()) continue;
                        const cols = lines[i].split(sep).map(c => c.trim().replace(/^"|"$/g, ''));
                        if (cols[0]) {
                            map.set(cols[0], {
                                Material: cols[0],
                                Description: cols[1] || '',
                                Quantity: parseFloat(cols[4]) || 0
                            });
                        }
                    }
                    resolve(map);
                };
                reader.readAsText(file, 'UTF-8');
            });

            showNotif("Đang phân tích và đối chiếu 2 kỳ...");
            const mapA = await parseFileAsync(fileA);
            const mapB = await parseFileAsync(fileB);

            let results = [];
            let allKeys = new Set([...mapA.keys(), ...mapB.keys()]);

            allKeys.forEach(mat => {
                const itemA = mapA.get(mat) || { Quantity: 0, Description: '' };
                const itemB = mapB.get(mat) || { Quantity: 0, Description: itemA.Description };
                const diff = itemB.Quantity - itemA.Quantity;
                
                let status = "Không đổi";
                if (!mapA.has(mat)) status = "Mã mới xuất hiện";
                else if (!mapB.has(mat)) status = "Mã đã mất (Hết tồn)";
                else if (diff > 0) status = "Tăng tồn kho";
                else if (diff < 0) status = "Giảm tồn kho";

                results.push({
                    Material: mat,
                    Description: itemB.Description || itemA.Description,
                    QtyA: itemA.Quantity,
                    QtyB: itemB.Quantity,
                    Diff: diff,
                    Status: status
                });
            });

            const tbody = document.querySelector('#comp-table tbody');
            tbody.innerHTML = '';
            results.forEach(r => {
                tbody.innerHTML += `<tr>
                    <td><b>${escapeHtml(r.Material)}</b></td>
                    <td>${escapeHtml(r.Description)}</td>
                    <td>${r.QtyA}</td>
                    <td>${r.QtyB}</td>
                    <td class="${r.Diff > 0 ? 'text-success' : (r.Diff < 0 ? 'text-danger' : '')}">${r.Diff > 0 ? '+' + r.Diff : r.Diff}</td>
                    <td><span class="badge">${r.Status}</span></td>
                </tr>`;
            });
            showNotif(`Đối chiếu hoàn tất! Tổng số ${results.length} bản ghi.`);
        }

        function initInventoryFromSAP() {
            if (sapData.length === 0) { showNotif("Vui lòng nạp dữ liệu SAP ở Tab Tra cứu trước!"); return; }
            inventoryData = sapData.map(r => ({ ...r, actualQty: r.Quantity, status: 'Chưa kiểm' }));

        }

        function renderInventoryTable() {
            const tbody = document.querySelector('#inv-table tbody');
            tbody.innerHTML = '';
            inventoryData.forEach((r, idx) => {
                const diff = r.actualQty - r.Quantity;
                tbody.innerHTML += `<tr>
                    <td><b>${escapeHtml(r.Material)}</b></td>
                    <td>${escapeHtml(r.Description)}</td>
                    <td>${escapeHtml(r.Location)}</td>
                    <td>${r.Quantity}</td>
                    <td><input type="number" value="${r.actualQty}" style="width:80px;" onchange="updateActualQty(${idx}, this.value)"></td>
                    <td class="${diff !== 0 ? 'text-danger' : 'text-success'}">${diff}</td>
                    <td><span class="badge">${r.status}</span></td>
                </tr>`;
            });
        }

        function updateActualQty(idx, val) {
            inventoryData[idx].actualQty = parseFloat(val) || 0;
            const diff = inventoryData[idx].actualQty - inventoryData[idx].Quantity;
            if (diff === 0) inventoryData[idx].status = 'Khớp';
            else if (diff > 0) inventoryData[idx].status = 'Thừa';
            else inventoryData[idx].status = 'Thiếu';
            renderInventoryTable();
        }

        function handleBarcode(e) {
            if (e.key === 'Enter') {
                const code = e.target.value.trim();
                const found = inventoryData.find(r => r.Material === code);
                if (found) {
                    found.actualQty += 1;
                    updateActualQty(inventoryData.indexOf(found), found.actualQty);
                    showNotif(`Đã quét mã vạch: ${code}`);
                } else {
                    showNotif(`Cảnh báo: Không tìm thấy Material ${code} trong danh sách!`);
                }
                e.target.value = '';
            }
        }

        async function executeFileSearch() {
            if (cachedFiles.length === 0) { alert("Vui lòng chọn thư mục trước khi tìm kiếm!"); return; }

            const nameKw = document.getElementById('searchName').value.toLowerCase().trim();
            const extKw = document.getElementById('searchExt').value.toLowerCase().trim().replace(/^\./, '');
            const contentKw = document.getElementById('searchContent').value.toLowerCase().trim();

            const statusEl = document.getElementById('file-search-status');
            const tbody = document.getElementById('fileResultBody');
            tbody.innerHTML = `<tr><td colspan="5" class="text-center">Đang quét và tìm kiếm tệp...</td></tr>`;
            statusEl.innerText = "Đang xử lý...";

            let results = [];
            for (let file of cachedFiles) {
                let fileName = file.name;
                let fileExt = fileName.split('.').pop().toLowerCase();
                let relativePath = file.webkitRelativePath || file.name;

                let matchName = !nameKw || fileName.toLowerCase().includes(nameKw);
                let matchExt = !extKw || fileExt === extKw;
                if (!matchName || !matchExt) continue;

                let matchType = [];
                if (nameKw) matchType.push("Tên file");
                if (extKw) matchType.push("Định dạng");

                let matchContent = true;
                if (contentKw) {
                    if (file.size < 5 * 1024 * 1024 && ['txt','csv','log','json','html','htm','xml','js','css','md'].includes(fileExt)) {
                        try {
                            const textContent = await file.text();
                            if (textContent.toLowerCase().includes(contentKw)) {
                                matchType.push("Nội dung");
                            } else { matchContent = false; }
                        } catch (err) { matchContent = false; }
                    } else { matchContent = false; }
                }

                if (matchName && matchExt && matchContent) {
                    results.push({
                        name: fileName, path: relativePath, ext: fileExt,
                        size: formatBytes(file.size), match: matchType.join(" + ")
                    });
                }
            }

            tbody.innerHTML = '';
            if (results.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="text-center">Không tìm thấy tệp nào phù hợp với điều kiện.</td></tr>`;
            } else {
                results.forEach(r => {
                    tbody.innerHTML += `<tr>
                        <td><b>${escapeHtml(r.name)}</b></td>
                        <td><small style="color: #666;">${escapeHtml(r.path)}</small></td>
                        <td><span class="badge">.${escapeHtml(r.ext)}</span></td>
                        <td>${r.size}</td>
                        <td style="color: #0288d1; font-style: italic;">${r.match}</td>
                    </tr>`;
                });
            }
            statusEl.innerText = `Tìm thấy ${results.length} kết quả phù hợp trên tổng số ${cachedFiles.length} tệp.`;
        }

        }

        // Helpers
        function downloadBlob(content, filename, contentType) {
            const blob = new Blob(["\uFEFF" + content], { type: contentType });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url; a.download = filename; a.click();
            URL.revokeObjectURL(url);
        }

        function formatBytes(bytes, decimals = 2) {
            if (bytes === 0) return '0 Bytes';
            const k = 1024, dm = decimals < 0 ? 0 : decimals, sizes = ['Bytes', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
        }

        function escapeHtml(str) {
            return String(str).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
